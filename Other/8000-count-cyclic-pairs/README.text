# TEST CASES

tests = [
    # Example 1
    ([1001, 1100, 110, 11], 1),

    # 四個元素屬於同一組
    ([1234, 3412, 4123, 1234], 6),

    # 長度不同不能配對
    ([10, 1, 100], 0),

    # 兩個不同的 rotation groups
    ([12, 21, 34, 43, 56], 2),

    # rotation 中可能出現 leading zero
    ([101, 110, 11], 1),

    # 三個相同元素
    ([0, 0, 0], 3),

    # 只有前兩個互為 rotation
    ([1212, 2121, 1221], 1),

    # 兩組各有三個元素
    ([123, 231, 312, 132, 321, 213], 6),

    # 同一組有重複數值
    ([1001, 1100, 11, 110, 1001], 3),

    # 沒有任何合法 pair
    ([1, 22, 333, 4444], 0),

    # 單一元素
    ([12345], 0),
]


solution = Solution()

for i, (nums, expected) in enumerate(tests, 1):
    actual = solution.countCyclicPairs(nums)

    if actual == expected:
        print(f"Test {i}: PASS")
    else:
        print(f"Test {i}: FAIL")
        print(f"  nums     = {nums}")
        print(f"  expected = {expected}")
        print(f"  actual   = {actual}")
